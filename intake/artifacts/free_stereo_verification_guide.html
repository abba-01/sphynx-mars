def find_sphinx_region(self):
"""Automatically detect sphinx-like structure"""
# Use edge detection and contour analysis
edges = cv2.Canny(self.left_img, 50, 150)
contours, _ = cv2.findContours(edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)

# Filter contours by size and shape
sphinx_candidates = []
for contour in contours:
area = cv2.contourArea(contour)
if 1000 < area < 50000: # Adjust based on image scale
sphinx_candidates.append(contour)

return sphinx_candidates

def measure_dimensions(self, contour, scale_factor):
"""Extract precise measurements from contour"""
# Fit bounding rectangle
rect = cv2.minAreaRect(contour)
width, height = rect[1]

# Convert pixels to meters using scale factor
width_meters = width * scale_factor
height_meters = height * scale_factor

return {
'width': width_meters,
'height': height_meters,
'area': cv2.contourArea(contour) * scale_factor**2,
'center': rect[0]
}

def calculate_distances(self, sphinx_center, twin_peaks_coords):
"""Calculate distances to Twin Peaks"""
distances = []
for peak in twin_peaks_coords:
dist = np.sqrt((sphinx_center[0] - peak[0])**2 +
(sphinx_center[1] - peak[1])**2)
distances.append(dist)
return distances

def extract_sphinx_measurements():
"""Main measurement extraction function"""

# Load processed data
analyzer = MarsSphinxAnalyzer(
'data/processed/left_ortho.tif',
'data/processed/right_ortho.tif',
'data/processed/stereo-DEM.tif'
)

# Define scale factor (meters per pixel)
# This comes from ISIS processing metadata
scale_factor = 0.5 # Example: 0.5 meters per pixel

# Find sphinx structure
sphinx_contours = analyzer.find_sphinx_region()

# Process each candidate
results = []
for i, contour in enumerate(sphinx_contours):
measurements = analyzer.measure_dimensions(contour, scale_factor)
measurements['id'] = f'candidate_{i}'
results.append(measurements)

# Twin Peaks coordinates (manually identified)
twin_peaks = [(x1, y1), (x2, y2)] # Pixel coordinates

# Calculate distances
for result in results:
distances = analyzer.calculate_distances(
result['center'], twin_peaks
)
result['distance_to_north_twin'] = distances[0] * scale_factor
result['distance_to_south_twin'] = distances[1] * scale_factor

return results

def calculate_orientation(contour):
"""Calculate orientation bearing of sphinx"""
# Fit ellipse to contour
ellipse = cv2.fitEllipse(contour)
angle = ellipse[2] # Orientation angle

# Convert to compass bearing
bearing = (90 - angle) % 360
return bearing

def generate_3d_model(dem_path, output_path):
"""Create 3D model from DEM"""
dem = cv2.imread(dem_path, cv2.IMREAD_ANYDEPTH)

# Create point cloud
points = []
for y in range(dem.shape[0]):
for x in range(dem.shape[1]):
z = dem[y, x]
if z > 0: # Valid elevation
points.append([x, y, z])

# Save as PLY file
point_cloud = o3d.geometry.PointCloud()
point_cloud.points = o3d.utility.Vector3dVector(points)
o3d.io.write_point_cloud(output_path, point_cloud)

return len(points)

📈 Statistical Analysis Script

import pandas as pd
import numpy as np
from scipy import stats
import matplotlib.pyplot as plt

def statistical_analysis(measurements):
"""Comprehensive statistical analysis of results"""

# Create DataFrame
df = pd.DataFrame(measurements)

# Calculate confidence intervals
def confidence_interval(data, confidence=0.95):
n = len(data)
mean = np.mean(data)
std_err = stats.sem(data)
interval = std_err * stats.t.ppf((1 + confidence) / 2, n-1)
return mean, interval

# Height analysis
heights = df['height'].values
height_mean, height_ci = confidence_interval(heights)

# Width analysis
widths = df['width'].values
width_mean, width_ci = confidence_interval(widths)

# Distance analysis
north_distances = df['distance_to_north_twin'].values
north_mean, north_ci = confidence_interval(north_distances)

results = {
'sphinx_height': {
'mean': height_mean,
'confidence_interval': height_ci,
'accuracy_class': 'high' if height_ci < height_mean * 0.05 else 'medium'
},
'sphinx_width': {
'mean': width_mean,
'confidence_interval': width_ci,
'accuracy_class': 'high' if width_ci < width_mean * 0.05 else 'medium'
},
'distance_validation': {
'mean': north_mean,
'confidence_interval': north_ci,
'known_value': 853.44, # 2800 feet in meters
'error_percent': abs(north_mean - 853.44) / 853.44 * 100
}
}

return results
