Extract precise measurements using computer vision techniques.

import cv2 import numpy as np import matplotlib.pyplot as plt from scipy.spatial.distance import cdist import open3d as o3d class MarsSphinxAnalyzer: def __init__(self, left_img_path, right_img_path, dem_path): self.left_img = cv2.imread(left_img_path, cv2.IMREAD_GRAYSCALE) self.right_img = cv2.imread(right_img_path, cv2.IMREAD_GRAYSCALE) self.dem = cv2.imread(dem_path, cv2.IMREAD_ANYDEPTH) def find_sphinx_region(self): """Detect sphinx-like structures using edge detection""" edges = cv2.Canny(self.left_img, 50, 150) contours, _ = cv2.findContours(edges, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE) # Filter by area and shape sphinx_candidates = [] for contour in contours: area = cv2.contourArea(contour) if 1000 < area < 50000: sphinx_candidates.append(contour) return sphinx_candidates def measure_dimensions(self, contour, scale_factor): """Extract measurements from detected contour""" rect = cv2.minAreaRect(contour) width, height = rect[1] return { 'width_meters': width * scale_factor, 'height_meters': height * scale_factor, 'area_sq_meters': cv2.contourArea(contour) * scale_factor**2, 'center_coords': rect[0] }
def calculate_twin_peaks_distance(sphinx_center, twin_peaks_coords, scale_factor): """Calculate distances to Twin Peaks for validation""" distances = [] for peak in twin_peaks_coords: pixel_dist = np.sqrt((sphinx_center[0] - peak[0])**2 + (sphinx_center[1] - peak[1])**2) meter_dist = pixel_dist * scale_factor distances.append(meter_dist) return distances def generate_3d_model(dem_path, output_path): """Create 3D point cloud from DEM""" dem = cv2.imread(dem_path, cv2.IMREAD_ANYDEPTH) points = [] for y in range(dem.shape[0]): for x in range(dem.shape[1]): z = dem[y, x] if z > 0: # Valid elevation points.append([x, y, z]) # Export as PLY file point_cloud = o3d.geometry.PointCloud() point_cloud.points = o3d.utility.Vector3dVector(points) o3d.io.write_point_cloud(output_path, point_cloud) return len(points)
